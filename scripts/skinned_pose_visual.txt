# experiment 정점 실증 시나리오 (I5-D34a/b)
#
# 사용법:
#   pwsh Tools\regression\verify-experiment-vertex-live.ps1
#   (경로 치환은 그 스크립트가 한다 — MBC9로 A/B 스위치는 은퇴)
#
# 관측 지점이 dx12.live가 아니라 dx12.scene인 이유: --script 헤드리스에서
# 라이브는 프레임을 완성하지 않는다(렌더 0프레임 실측 — present가 없다).
# dx12.scene 오프라인 하네스는 같은 씬 프록시를 자체 캐시로 그리고 커버리지·
# 밝기 단정까지 갖고 있어, experiment 버퍼로 그린 그림이 통째로 틀리면
# 하네스가 붉는다 — 계수 관측에 부분적 픽셀 판정이 얹힌다.

scene.switch __SCENE__

wait 60

# I5-D34b: 스킨 메시(Gunner — 스킨 마스크 68B, BLENDINDICES uint)를 같은
# 씬에 배치한다. FT 프리미티브는 전부 정적 core라 스킨 레이아웃 축이
# 이 배치 없이는 단 한 번도 돌지 않는다.
model.load __SKINNED_MODEL__
wait 120
model.place Gunner_F_Mythic
wait 60
# 원점 그대로 두면 캐릭터가 씬 카메라를 삼켜 커버리지가 전면(65536)이 되고
# 하네스의 라이팅 단정("광원이 셰이더에 닿지 않는다")이 무너진다. CLI
# transform은 라이브 프록시에 전파되지 않으므로(실측 — 스케일 0.01로도
# 커버리지 불변), 저작값으로 기록한 뒤 저장·재로드로 반영한다 — 로드
# 경로는 저작값으로 프록시를 새로 만든다.
object.transform Gunner_F_Mythic 1.5 0 0 0 0 0 0.3 0.3 0.3
wait 30
# 애니메이션 틱은 실시간(deltaTime) 기반이라 포즈가 실행마다 다르다 — A/B
# 커버리지·밝기 동수 단정과 양립하지 않는다. 저장 전에 Animator를 꺼서
# 재로드 씬이 결정적 바인드 포즈로 그려지게 한다. 스킨 레이아웃 판정에는
# 손실이 없다 — 정점 페치(BLENDINDICES uint 조립)는 스키닝 분기와 무관하게
# PSO 레이아웃대로 일어난다. 스키닝 산술의 시각 판정은 비결정성 때문에 이
# 게이트 밖이다(계획서 한계 기록).
experiment.animpose __POSE_FRACTION__
wait 30
# I5-D34c: 오브젝트 하나를 투명으로 바꿔 forward 큐를 실제로 채운다 — FT는
# 전부 불투명(deferred)이라 이 한 줄 없이는 Forward 레이아웃 축이 실씬에서
# 단 한 번도 돌지 않는다. renderingMode는 재질 직렬화에 실리므로 저장·재로드
# 후에도 유지된다.
render.matmode P_Torus transparent
wait 30
# I5-D4e-2: 합성 이벤트·루프 오버라이드 주입 — 코퍼스 저작분 0이라 합성 없이는
# 이관(왕복·비오염·발화)이 원리적으로 판정 불가. 저장 직전에 심어 아래
# 재로드에서 verify가 왕복을 잰다.
experiment.animevent seed
wait 10
# I5-D5a: 합성 Foliage 저작(같은 이유 — 코퍼스 Foliage 저작분 0). 타입 저작
# 경로(AddFoliageType)와 자산 게시까지 심어, 재로드 verify가 postLoad 재해석
# 바인딩과 프록시 DrawSource 핸들을 잰다.
experiment.foliage seed __FOLIAGE_DIR__ "__SKINNED_MODEL__"
wait 30
# I5-D5c1: 합성 새 정본 재질 저작 — 코퍼스에 shaderAssetId/ref 표기가 0건이라
# (실측) 저작 경로가 실자산에서 한 번도 돈 적이 없다. seed가 자산을 게시하고
# renderer를 base에 링크해, 저장이 ref 표기를 내고 재로드가 병행 표현을 채운다.
experiment.matruntime seed "__MATERIAL_DIR__" "__TEXTURE_PATH__"
wait 30
scene.save __SAVED_SCENE__
wait 30
scene.switch __SAVED_SCENE__
wait 120

# I5-D4e-1: 재생 팔레트 패리티 — 재로드 뒤에 돌려 postLoad의 experiment
# 재생 바인딩(EnsureExperimentAnimationBinding)까지 함께 검증한다. 라이브
# 틱은 실시간이라 비결정 — 이 명령이 결정적 표본으로 같은 제품 함수를 잰다.
experiment.animtick
wait 10
# I5-D4e-2: 왕복·비오염·발화 판정(위 seed의 대구).
experiment.animevent verify
wait 10
# I5-D4e-3: 본 이름 해석 창구 전수 A/B + AvatarMask 트리 생성 A/B.
experiment.boneresolve
wait 10
experiment.animmask
wait 10
# I5-D5b: 에디터 실소비 창구 전수 A/B(클립 열거·이름, 메시 존재 가드).
experiment.editorsurface
wait 10
# I2-E: 임베디드 텍스처 신원 — 모델 sidecar의 subAssets가 소스 로드 경로에도
# 실리는가(예전에는 property가 통째로 생략됐다 — 실측 textureProps 0).
wait 10
# I5-D4f-1: 바운드 축 — 정점 시공 절단(legacyVertices)·바운드 생존
# (degenerate)·두 유도의 값 동수(digest, ps1이 on/off 대조).
wait 10
# I5-D5c1: 재질 병행 표현 A/B + 왕복 손실 실측.
experiment.matruntime
wait 10
# I5-D5c3: 편집 반영 축 — 실물 편집 창구를 태우고 저작 인스턴스가 따라오는가.
experiment.matruntime edit
wait 10
# I5-D5a: Foliage 핸들 왕복·프록시 DrawSource 판정(위 seed의 대구 — 스크립트가
# 모드 인자를 치환한다).
experiment.foliage verify __FOLIAGE_MODE__
wait 10

dx12.scene

quit
