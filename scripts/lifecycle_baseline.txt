# 생명주기 기준선 시나리오 (PHASE 9-0)
#
# 사용법:
#   CreatorEditor.exe --script ..\..\scripts\lifecycle_baseline.txt
#   (Tools\regression\verify-lifecycle-baseline.ps1 이 이 파일을 부른다)
#
# 목적:
#   로드 → 재생 → 전환 → 종료 한 바퀴 동안 생명주기가 누구를 어떤 순서로 부르는지
#   받아 적는다. PHASE 9-1~9-3이 그 기구를 통째로 바꾸므로, 교체 후 같은 시나리오로
#   다시 받아 두 파일을 비교하는 것이 기능 동등성 판정 방법이다.
#
# 왜 왕복(A→B→A)인가:
#   편도만 재면 "새 씬에서 안 불렸다"와 "이전 씬이 정리되지 않았다"를 구분할 수 없다.
#   씬 전환의 OnDestroy 대칭성은 왕복에서만 드러난다.
#
# 씬 경로:
#   {{SCENE_A}} / {{SCENE_B}}는 verify-lifecycle-baseline.ps1이 실행 직전에
#   절대 경로로 치환한다. scene.switch가 절대 경로를 받기 때문이고, 절대 경로를
#   여기 박아 두면 다른 사람의 작업 폴더에서 조용히 실패한다.
#   기본값은 -SceneA/-SceneB로 바꿀 수 있다.
#
# ── 두 씬을 이 시나리오가 직접 만든다 (2026-08-21) ──
#
#   예전에는 저작 자산 `FT_Material`(3D) / `UITestScene`(UI)을 열었다. 둘 다
#   §0.05의 폐기 대상이었고, `UITestScene`은 아예 `.gitignore`가 무시하는 파일이라
#   이 게이트는 **이 기계에서만 돌고 있었다.**
#
#   재현 목표는 옛 기준선이 담던 컴포넌트 5종이다(실측 분포):
#     MeshRenderer 132 · LightComponent 26 · CameraComponent 26 · ImageComponent 8 · Canvas 8
#   그래서 A에는 카메라·라이트·메시를, B에는 캔버스·이미지·텍스트를 둔다.
#   구조가 다른 둘이어야 한다는 원래 요건(아래)은 그대로다.
#
#   ★ 저작은 `lifecycle.trace clear`보다 **앞**이어야 한다. 뒤에 두면 만드는
#   과정의 Awake가 기록에 섞여, 무엇이 왕복의 결과인지 구분되지 않는다.
#
#   ★ 저작을 마치면 **빈 씬으로 한 번 비운다**(LifecycleScratch). 안 비우면 첫
#   `scene.switch {{SCENE_A}}`가 방금 저작하던 씬을 통째로 파괴하며 그 기록이
#   첫 줄들을 차지한다. 옛 기준선의 첫 줄이 정확히 그 모양이었다(기본 씬의
#   `Main Camera` OnEndSimulation) — 우연히 결정론적이었을 뿐 의도가 아니었다.
#
# 틱 프레임 예산:
#   3으로 두는 이유는 한 프레임 안의 순서가 알고 싶은 것이지 반복 횟수가
#   아니기 때문이다. 늘리면 파일만 커지고 diff가 어려워진다.

# --- 기록 시작 ---
#
# 첫 씬 로드보다 앞이어야 한다. 로드 뒤에 켜면 그 씬의 Awake가 빠진 채 나중에
# OnDestroy만 잡혀 두 수가 어긋나고, 그 어긋남이 결함인지 계측 구멍인지
# 구분되지 않는다.
#
# 예산을 0으로 주는 것은 '수명 단계만'이라는 뜻이다. 틱 단계는 재생이 안정된
# 뒤에 따로 연다 — 아래 주석 참고.
# --- 씬 저작 (기록 시작 전) ---
#
# 씬 B — UI. Canvas·Image·Text가 Update 디스패치에 잡히는 쪽이다.
scene.new LifecycleUI
wait 40
object.create UICanvas Canvas
component.add UICanvas Canvas
wait 20
object.create UIImageA UI
component.add UIImageA ImageComponent
object.parent UIImageA UICanvas
object.create UIImageB UI
component.add UIImageB ImageComponent
object.parent UIImageB UICanvas
object.create UILabel UI
component.add UILabel TextComponent
object.parent UILabel UICanvas
wait 30
scene.save {{SCENE_B}}
wait 40

# 씬 A — 3D. 카메라·라이트·메시. lifecycle.stress가 8개를 집으므로 메시를
# 넉넉히 둔다(파괴 대상이 모자라면 스트레스가 아무것도 재지 못한다).
scene.new Lifecycle3D
wait 40
object.create MainCamera
component.add MainCamera CameraComponent
object.create SunLight
component.add SunLight LightComponent
object.create FillLight
component.add FillLight LightComponent
wait 20
object.create Mesh01
component.add Mesh01 MeshRenderer
object.create Mesh02
component.add Mesh02 MeshRenderer
object.create Mesh03
component.add Mesh03 MeshRenderer
object.create Mesh04
component.add Mesh04 MeshRenderer
object.create Mesh05
component.add Mesh05 MeshRenderer
object.create Mesh06
component.add Mesh06 MeshRenderer
object.create Mesh07
component.add Mesh07 MeshRenderer
object.create Mesh08
component.add Mesh08 MeshRenderer
object.create Mesh09
component.add Mesh09 MeshRenderer
object.create Mesh10
component.add Mesh10 MeshRenderer
object.create Mesh11
component.add Mesh11 MeshRenderer
object.create Mesh12
component.add Mesh12 MeshRenderer
wait 40
scene.save {{SCENE_A}}
wait 40

# 저작하던 씬을 비운다 — 위 머리말의 두 번째 ★.
scene.new LifecycleScratch
wait 60

lifecycle.trace clear
lifecycle.trace on 0

# 디스패치 경로 선택(PHASE 9-1). 검증 스크립트가 -Registry 여부에 따라 채운다.
# 첫 scene.switch보다 앞이어야 한다 — 전환은 씬 경계에서만 안전하다.
{{REGISTRY_SETUP}}

# --- 로드 ---
scene.switch {{SCENE_A}}
wait 60

# --- 재생 진입: Awake/OnEnable/Start 순서가 여기 담긴다 ---
play
wait 120

# 여기서 틱 예산을 연다.
#
# 처음에는 play 앞에서 열었는데, play는 다음 프레임에 처리되므로 예산 3프레임이
# 에디터 모드 프레임에 그대로 소진됐다 — 재생 중 Update를 한 건도 담지 못한 채
# 기준선이 떠졌다(실측: 프레임 0·1·2의 LightComponent·CameraComponent뿐).
# 9-1이 교체하는 것이 바로 그 Update 디스패치라 그 구멍은 대조를 무의미하게 만든다.
lifecycle.trace on 3
wait 10
lifecycle.trace status

# --- 파괴 경로: OnDestroy 순서와 파괴 중 순회 안전성 ---
lifecycle.stress destroy 8
wait 60
lifecycle.stress churn 8
wait 60

# --- 전환: 이전 씬의 정리와 새 씬의 초기화가 겹치는 구간 ---
scene.switch {{SCENE_B}}
wait 120

# 두 번째 씬에서 틱 예산을 한 번 더 연다.
#
# 첫 씬만으로는 Update를 구현한 컴포넌트 13종 중 두 종(Light·Camera)밖에 담기지
# 않는다 — 3D 씬에 UI가 없기 때문이다. UI 씬에서 Canvas·Image·Text 계열이 함께
# 잡혀야 9-1이 교체하는 Update 디스패치의 순서 계약을 실제로 대조할 수 있다.
lifecycle.trace on 3
wait 10

scene.switch {{SCENE_A}}
wait 120

# --- 정지 ---
stop
wait 60

# --- 기록 저장 ---
lifecycle.trace off
lifecycle.dump lifecycle_trace.tsv
scene.dump lifecycle-baseline-final
gpu.census lifecycle-baseline-final
# 관리 힙도 함께 남긴다(9-7). 9-3의 전/후 비교에서 재설계 효과와 GC 효과를
# 분리 판정하려면 두 수치가 같은 실행에 있어야 한다.
gc.stats lifecycle-baseline-final
log.flush
quit
