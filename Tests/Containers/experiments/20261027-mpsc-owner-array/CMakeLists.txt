cmake_minimum_required(VERSION 3.28)
project(MpscCursorControl LANGUAGES CXX)
file(GLOB_RECURSE units CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/modules/*.ixx")
add_library(ce_modules STATIC)
target_compile_features(ce_modules PUBLIC cxx_std_23)
target_compile_options(ce_modules PRIVATE /permissive- /EHsc /utf-8)
target_sources(ce_modules PUBLIC FILE_SET CXX_MODULES BASE_DIRS "${CMAKE_SOURCE_DIR}/modules" FILES ${units})
add_executable(mpsc_control bench.cpp)
target_compile_features(mpsc_control PRIVATE cxx_std_23)
target_include_directories(mpsc_control PRIVATE "${CMAKE_SOURCE_DIR}/../../tests")
target_compile_options(mpsc_control PRIVATE /permissive- /EHsc /utf-8 /Ob3)
target_link_libraries(mpsc_control PRIVATE ce_modules)
