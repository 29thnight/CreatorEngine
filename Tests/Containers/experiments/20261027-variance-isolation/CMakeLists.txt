cmake_minimum_required(VERSION 3.25)
project(variance_isolation LANGUAGES CXX)
add_executable(variance_context main.cpp context.cpp)
target_include_directories(variance_context PRIVATE ../../../../Engine/Containers/include)
target_link_libraries(variance_context PRIVATE pdh psapi)
add_executable(controlled_process controlled_process.cpp)
add_executable(variance_load variance_load.cpp)
foreach(target variance_context controlled_process variance_load)
 target_compile_features(${target} PRIVATE cxx_std_23)
 target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
endforeach()
add_executable(controlled_cpu controlled_cpu.cpp)
add_executable(stream_load stream_load.cpp)
add_executable(variance_cpu_context cpu-context/main.cpp cpu-context/context.cpp)
target_include_directories(variance_cpu_context PRIVATE ../../../../Engine/Containers/include)
target_link_libraries(variance_cpu_context PRIVATE pdh psapi)
foreach(target controlled_cpu stream_load variance_cpu_context)
 target_compile_features(${target} PRIVATE cxx_std_23)
 target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
endforeach()
