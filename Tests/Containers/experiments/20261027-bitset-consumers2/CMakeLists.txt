cmake_minimum_required(VERSION 3.28)
project(BitsetConsumerBench LANGUAGES CXX)
get_filename_component(CE_MODULE_DIR "${CMAKE_SOURCE_DIR}/../../../../Engine/Containers/modules" ABSOLUTE)
file(GLOB_RECURSE units CONFIGURE_DEPENDS "${CE_MODULE_DIR}/*.ixx")
add_library(ce_modules STATIC)
target_compile_features(ce_modules PUBLIC cxx_std_23)
target_compile_options(ce_modules PRIVATE /permissive- /EHsc /utf-8)
target_sources(ce_modules PUBLIC FILE_SET CXX_MODULES BASE_DIRS ${CE_MODULE_DIR} FILES ${units})
add_executable(bitset_consumer bench.cpp)
target_compile_features(bitset_consumer PRIVATE cxx_std_23)
target_compile_options(bitset_consumer PRIVATE /permissive- /EHsc /utf-8 /Ob3)
target_link_libraries(bitset_consumer PRIVATE ce_modules)
