cmake_minimum_required(VERSION 3.28)
project(FaaTeardownControl LANGUAGES CXX)
file(GLOB_RECURSE units CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/modules/*.ixx")
add_library(ce_modules STATIC)
target_compile_features(ce_modules PUBLIC cxx_std_23)
target_compile_options(ce_modules PRIVATE /permissive- /EHsc /utf-8)
target_sources(ce_modules PUBLIC FILE_SET CXX_MODULES BASE_DIRS "${CMAKE_SOURCE_DIR}/modules" FILES ${units})
add_executable(faa_control bench.cpp)
target_compile_features(faa_control PRIVATE cxx_std_23)
target_compile_definitions(faa_control PRIVATE CE_USE_FAA=1)
target_compile_options(faa_control PRIVATE /permissive- /EHsc /utf-8 /Ob3)
target_link_libraries(faa_control PRIVATE ce_modules)
