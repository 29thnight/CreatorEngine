cmake_minimum_required(VERSION 3.25)
project(small_array_context LANGUAGES CXX)
add_executable(array_context main.cpp context.cpp)
target_compile_features(array_context PRIVATE cxx_std_23)
target_include_directories(array_context PRIVATE ../../include)
target_compile_options(array_context PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
target_link_libraries(array_context PRIVATE pdh psapi)
add_executable(smt_guard smt_guard.cpp)
target_compile_features(smt_guard PRIVATE cxx_std_23)
target_compile_options(smt_guard PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
