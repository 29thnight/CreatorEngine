cmake_minimum_required(VERSION 3.25)
project(batch_window LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
find_package(Threads REQUIRED)
foreach(pair core expanded)
 if(pair STREQUAL "core")
  set(src main.cpp)
 else()
  set(src library_benchmark.cpp)
 endif()
 add_executable(${pair} ${src})
 target_include_directories(${pair} PRIVATE ../../include)
 target_link_libraries(${pair} PRIVATE Threads::Threads)
 target_compile_options(${pair} PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
endforeach()
