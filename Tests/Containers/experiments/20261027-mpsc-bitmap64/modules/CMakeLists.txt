function(module_validation target source)
 set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/tests/${source}")
 file(READ "${PROJECT_SOURCE_DIR}/tests/${source}" text)
 string(REGEX REPLACE "#include <ce/[^>]+>" "// API supplied by named module" text "${text}")
 set(output "${CMAKE_CURRENT_BINARY_DIR}/${target}.cpp")
 file(WRITE "${output}" "#include \"${PROJECT_SOURCE_DIR}/tests/modules/prelude.h\"\nimport Creator.Containers;\n${text}")
 add_executable(${target} "${output}")
 target_include_directories(${target} PRIVATE "${PROJECT_SOURCE_DIR}/tests")
 target_link_libraries(${target} PRIVATE CreatorContainers::Modules Threads::Threads)
 if(MSVC)
  target_compile_options(${target} PRIVATE /W4 /WX)
 endif()
endfunction()
