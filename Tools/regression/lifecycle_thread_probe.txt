# LC5-b 스레드 경계 픽스처 (ScriptLifecycleContractHardeningPlan · LC5)
#
# await가 재개되는 스레드를 받아 적는다. 판정은 verify-lifecycle-thread.ps1이 한다.
#
# ── 왜 인스턴스가 하나뿐인가 ──
#
# 다른 생명주기 픽스처와 달리 대조군을 옆에 두지 않는다. 여기서 갈라야 하는 두
# 값(지원 경로 재개 · 외부 await 재개)이 이미 같은 인스턴스의 같은 루틴 안에
# 나란히 있기 때문이다. 인스턴스를 늘리면 스레드 풀 사정이 섞여 오히려 축이
# 흐려진다.
#
# ── 왜 play가 필요한가 ──
#
# C# 스크립트는 재생 중에만 생명주기를 받는다(2026-09-05, 편집/재생 경계 고침).
# 편집 모드에서는 인스턴스만 만들어지고 훅은 한 건도 오지 않는다.

wait 60
scene.new LifecycleThread
wait 30

object.create ExternalAwait
script.add ExternalAwait ExternalAwaitProbe
wait 10

# 워커 몸통에서 엔진 API를 부른다 (LC5-c). 위 픽스처가 재는 "await 뒤"와 달리
# 여기는 마셜링이 원리적으로 닿지 못하는 자리다.
object.create WorkerCall
script.add WorkerCall WorkerCallProbe
wait 60

play

# Scope.Delay(0.1초) → Task.Delay(50ms) 순이라 재개 둘이 다 끝나려면 넉넉히 둔다.
wait 400

stop
wait 200

log.flush
quit
